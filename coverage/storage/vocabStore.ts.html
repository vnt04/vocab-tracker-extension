// ============================================================================
// src/storage/vocabStore.ts
// Storage service over chrome.storage.local. Single mutation path with an
// internal module-level write lock so overlapping Save clicks cannot clobber
// each other (read-modify-write is serialized).
// ============================================================================
import type { SaveResult, VocabularyItem } from './types';
 
const STORAGE_KEY = 'vocab.items.v1'; // single key holds the whole array
const MAX_LEN = 50; // length check AFTER trim+collapse
 
// Serializes overlapping writes so two rapid Save clicks can't read the same
// stale array and lose an update (last-write-wins data loss).
let writeChain: Promise<unknown> = Promise.resolve();
 
/** Run `fn` after any in-flight write completes, regardless of its outcome. */
function withLock<T>(fn: () => Promise<T>): Promise<T> {
  const run = writeChain.then(fn, fn);
  // Keep the chain alive even if `fn` rejects; swallow here so the next caller
  // isn't rejected by an unrelated prior failure.
  writeChain = run.catch(() => undefined);
  return run;
}
 
/** trim + collapse internal whitespace + lowercase. For COMPARISON ONLY — never persisted. */
export function normalizeText(text: string): string {
  return text.trim().replace(/\s+/g, ' ').toLowerCase();
}
 
/** `cleaned` = trim + collapse internal whitespace (casing preserved). */
function cleanText(text: string): string {
  return text.trim().replace(/\s+/g, ' ');
}
 
/** Read the single array under STORAGE_KEY. Returns [] on first run / parse miss / error. */
export async function getAll(): Promise<VocabularyItem[]> {
  try {
    const result = await chrome.storage.local.get(STORAGE_KEY);
    const raw = result[STORAGE_KEY];
    return Array.isArray(raw) ? (raw as VocabularyItem[]) : [];
  } catch {
    return [];
  }
}
 
/**
 * Case-insensitive upsert, race-safe via an internal module-level write lock.
 *  - cleaned empty            -> { status: 'invalid', reason: 'empty' }
 *  - cleaned.length > MAX_LEN -> { status: 'invalid', reason: 'too_long' }
 *  - no case-insensitive match -> create { id: crypto.randomUUID(), text: cleaned, count: 1, notes: '' }
 *  - match found              -> count+1, keep stored text casing + existing notes (immutable update)
 *  - storage throws           -> { status: 'error', message }
 */
export function saveOrIncrement(text: string): Promise<SaveResult> {
  const cleaned = cleanText(text);
 
  if (cleaned.length === 0) {
    return Promise.resolve({ status: 'invalid', reason: 'empty' });
  }
  if (cleaned.length > MAX_LEN) {
    return Promise.resolve({ status: 'invalid', reason: 'too_long' });
  }
 
  return withLock(async () => {
    try {
      const items = await getAll();
      const key = normalizeText(cleaned);
      const index = items.findIndex((item) => normalizeText(item.text) === key);
 
      if (index === -1) {
        const item: VocabularyItem = {
          id: crypto.randomUUID(),
          text: cleaned,
          count: 1,
          notes: '',
        };
        const next = [...items, item];
        await chrome.storage.local.set({ [STORAGE_KEY]: next });
        return { status: 'created', item };
      }
 
      const existing = items[index];
      const item: VocabularyItem = { ...existing, count: existing.count + 1 };
      const next = items.map((current, i) => (i === index ? item : current));
      await chrome.storage.local.set({ [STORAGE_KEY]: next });
      return { status: 'incremented', item };
    } catch (error) {
      const message = error instanceof Error ? error.message : String(error);
      return { status: 'error', message };
    }
  });
}
